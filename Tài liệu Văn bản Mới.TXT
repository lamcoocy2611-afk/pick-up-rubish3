/* key storage structure: {0:true/false, 1:true/false,...}
   meaning: for question i, if correctIsTrue === true then choosing "Đúng" is correct.
   By default keys are empty until teacher sets them (edit mode). */
const STORAGE_KEY = 'caf_chain_tf_answers_v1';
let editMode = false;
const quizContainer = document.getElementById('quiz');

function loadSavedAnswers(){
  try{
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : {};
  }catch(e){return {}}
}

function saveAnswersObj(obj){
  localStorage.setItem(STORAGE_KEY, JSON.stringify(obj));
}

let correctMap = loadSavedAnswers(); // may be empty

function render(){
  quizContainer.innerHTML = '';
  questions.forEach((text, idx) => {
    const card = document.createElement('div');
    card.className = 'card';
    const qnum = document.createElement('div');
    qnum.innerHTML = <span class="qnum">${idx+1}.</span> <span class="editable">${text.replace(/\n/g,'<br>')}</span>;
    card.appendChild(qnum);

    const opts = document.createElement('div');
    opts.className = 'opts';

    const trueLabel = document.createElement('label');
    trueLabel.className = 'opt';
    trueLabel.innerHTML = <span>Đúng (True)</span><input type="radio" name="q${idx}" value="true">;
    const falseLabel = document.createElement('label');
    falseLabel.className = 'opt';
    falseLabel.innerHTML = <span>Sai (False)</span><input type="radio" name="q${idx}" value="false">;

    opts.appendChild(trueLabel);
    opts.appendChild(falseLabel);
    card.appendChild(opts);

    // editor controls
    const editorRow = document.createElement('div');
    editorRow.style.marginTop='8px';
    if(editMode){
      const btn = document.createElement('button');
      btn.className='toggleCorrect';
      const current = correctMap[idx];
      btn.textContent = (current===true) ? 'Set correct = Đúng' : (current===false) ? 'Set correct = Sai' : 'Set correct = —';
      btn.onclick = () => {
        // cycle: undefined -> true -> false -> undefined
        if(correctMap[idx] === undefined) correctMap[idx] = true;
        else if(correctMap[idx] === true) correctMap[idx] = false;
        else delete correctMap[idx];
        saveAnswersObj(correctMap);
        render();
      };
      editorRow.appendChild(btn);
      const hint = document.createElement('span');
      hint.className='small';
      hint.style.marginLeft='10px';
      hint.innerHTML = Current: ${ correctMap[idx]===true ? '<span class="correctMark">Đúng</span>' : correctMap[idx]===false ? '<span class="wrongMark">Sai</span>' : '<em>not set</em>' };
      editorRow.appendChild(hint);
    }

    card.appendChild(editorRow);
    quizContainer.appendChild(card);
  });
}

document.getElementById('toggleEdit').addEventListener('click', (e)=>{
  editMode = !editMode;
  e.target.textContent = Edit mode: ${editMode? 'ON':'OFF'};
  document.getElementById('saveAnswers').style.display = editMode ? 'inline-block' : 'inline-block';
  render();
});

document.getElementById('saveAnswers').addEventListener('click', ()=>{
  saveAnswersObj(correctMap);
  alert('Answers saved to browser storage. Học sinh mở lại trang sẽ thấy we use these answers.');
});

document.getElementById('resetSaved').addEventListener('click', ()=>{
  if(confirm('Reset saved answers?')) {
    correctMap = {};
    saveAnswersObj(correctMap);
    render();
    alert('Đã reset.');
  }
});

document.getElementById('check').addEventListener('click', ()=>{
  const total = questions.length;
  let correctCount = 0;
  const results = [];
  for(let i=0;i<total;i++){
    const radios = document.getElementsByName('q'+i);
    let chosen = null;
    for(const r of radios) if(r.checked) chosen = r.value;
    const key = correctMap[i]; // true => Đúng is correct; false => Sai is correct; undefined => teacher not set
    if(key === undefined){
      results.push({i, status:'no-key', chosen});
      continue;
    }
    if(chosen === null){
      results.push({i, status:'no-answer', chosen});
      continue;
    }
    const isCorrect = ( (chosen === 'true' && key === true) || (chosen === 'false' && key === false) );
    if(isCorrect) correctCount++;
    results.push({i, status: isCorrect ? 'ok' : 'wrong', chosen});
  }

  // show score and per-question feedback
  const scoreDiv = document.getElementById('score');
  const numKeySet = Object.keys(correctMap).length;
  scoreDiv.innerHTML = Score: ${correctCount} / ${ (numKeySet>0)? numKeySet : total };
  // render feedback boxes
  const cards = document.querySelectorAll('.card');
  results.forEach(res => {
    const card = cards[res.i];
    // remove old feedback
    let fb = card.querySelector('.result');
    if(fb) fb.remove();
    fb = document.createElement('div');
    fb.className = 'result';
    if(res.status === 'no-key'){
      fb.innerHTML = <div class="small">Chưa có đáp án chuẩn cho câu này — giáo viên chưa set (Edit mode).</div>;
      fb.style.background = 'rgba(255,255,255,0.02)';
      card.appendChild(fb);
    } else if(res.status === 'no-answer'){
      fb.innerHTML = <div class="small">Bạn chưa chọn Đúng/Sai cho câu này.</div>;
      fb.style.background = 'rgba(255,255,255,0.02)';
      card.appendChild(fb);
    } else if(res.status === 'ok'){
      fb.innerHTML = <div class="small correctMark">✅ Đúng</div>;
      fb.style.background = 'rgba(38,166,154,0.08)';
      card.appendChild(fb);
    } else {
      fb.innerHTML = <div class="small wrongMark">❌ Sai</div>;
      fb.style.background = 'rgba(255,107,107,0.06)';
      card.appendChild(fb);
    }
  });

});

document.getElementById('clear').addEventListener('click', ()=>{
  document.querySelectorAll('input[type=radio]').forEach(r=> r.checked=false);
  document.getElementById('score').textContent = '';
  document.querySelectorAll('.card .result').forEach(n=> n.remove());
});

// init
render();
</script>
